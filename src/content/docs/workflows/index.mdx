import requests as req
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters

# توکن ربات تلگرام
TOKEN = "7314937475:AAFaQcpZL9tX2Xqio7odoPKQw5Hd0hzwP88"
CHANNEL_ID = "@https://t.me/+4bICe3mjLyxkYTM0"  # نام کاربری کانال شما (با @)

# تابع شروع ربات
async def start(update: Update, context):
    await update.message.reply_text("سلام! لطفاً سوالت را بپرس.")

# بررسی عضویت کاربر در کانال
async def check_membership(update: Update, context):
    user_id = update.message.from_user.id
    status = await context.bot.get_chat_member(CHANNEL_ID, user_id)
    if status.status in ['member', 'administrator', 'creator']:
        return True
    else:
        await update.message.reply_text("برای استفاده از این ربات، لطفاً ابتدا در کانال عضو شوید.")
        return False

# پاسخ به سوالات کاربران
async def handle_message(update: Update, context):
    if await check_membership(update, context):
        question = update.message.text
        try:
            response = req.get(f"http://5.161.91.18/chat?text={question}")
            if response.status_code == 200:
                answer = response.json()  # پاسخ API شما
                await update.message.reply_text(answer)
            else:
                await update.message.reply_text("خطا در دریافت پاسخ از سرور.")
        except Exception as e:
            await update.message.reply_text(f"خطا: {str(e)}")

# شروع ربات تلگرام
def main():
    application = ApplicationBuilder().token(TOKEN).build()

    # هندلرهای ربات
    application.add_handler(CommandHandler("start", start))
    application.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))

    # شروع به فعالیت
    application.run_polling()

if __name__ == "__ad1__":
    main()
